    Definir intentos_fallidos Como Entero

    Definir CLAVE_CORRECTA, CODIGO_REQUERIDO Como Entero
    CLAVE_CORRECTA <- 2026
    CODIGO_REQUERIDO <- 1
    
    // Inicialización
    intentos_fallidos <- 0
    clave_acceso <- 0
    codigo_seguridad <- 0
    
    Escribir "=== PANEL DE ACCESO RESTRINGIDO ==="
    Escribir "Sistema de seguridad de alto nivel"
    Escribir ""

    Repetir
        // Verificar si se alcanzó el límite de intentos
        Si intentos_fallidos >= 3 Entonces
            Escribir "========================================"
            Escribir "¡ALERTA DE SEGURIDAD!"
            Escribir "Intento de intrusión detectado"
            Escribir "Sistema bloqueado permanentemente"
            Escribir "========================================"
            intentos_fallidos <- 3  // Asegurar salida del bucle
        Sino
            // Solicitar credenciales
            Escribir "--- Intento N°", intentos_fallidos + 1, " ---"
            Escribir "Ingrese clave de acceso (numérica):"
            Leer clave_acceso
            Escribir "Ingrese código de seguridad (1-3):"
            Leer codigo_seguridad
            
            // Validar credenciales
            Si clave_acceso = CLAVE_CORRECTA Y codigo_seguridad = CODIGO_REQUERIDO Entonces
                Escribir ""
                Escribir "========================================"
                Escribir "✔ ACCESO CONCEDIDO"
                Escribir "Bienvenido al sistema administrativo"
                Escribir "========================================"
                intentos_fallidos <- 3  // Salir del bucle
            Sino
                intentos_fallidos <- intentos_fallidos + 1
                Escribir "✘ CREDENCIALES INCORRECTAS"
                Escribir "Intentos fallidos: ", intentos_fallidos, "/3"
                
                // Mensajes de ayuda según el error
                Si clave_acceso <> CLAVE_CORRECTA Entonces
                    Escribir "   → La clave de acceso es incorrecta"
                FinSi
                Si codigo_seguridad <> CODIGO_REQUERIDO Entonces
                    Escribir "   → El código de seguridad debe ser 1"
                FinSi
                Escribir ""
            FinSi
        FinSi
        
    Hasta Que intentos_fallidos >= 3
    
    Si intentos_fallidos = 3 Y (clave_acceso <> CLAVE_CORRECTA O codigo_seguridad <> CODIGO_REQUERIDO) Entonces
        // Mensaje de bloqueo ya mostrado dentro del bucle
    FinSi
FinAlgoritmo
